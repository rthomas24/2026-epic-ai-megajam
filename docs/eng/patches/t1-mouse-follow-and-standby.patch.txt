+float ATDSpacePlayerController::ReticleFullDegrees() const
+{
+	// The angle off the camera's centre that the reticle shows at full deflection (the same projection as
+	// GetAimWorldDirectionForViewport, horizontally).
+	int32 SizeX = 0, SizeY = 0;
+	GetViewportSize(SizeX, SizeY);
+	if (SizeX <= 0 || SizeY <= 0) { SizeX = 1920; SizeY = 1080; }
+	float FOV = 90.f;
+	if (const ATDShip* Ship = Cast<ATDShip>(GetSpacecraft()))
+	{
+		if (Ship->IsCockpitView() && Ship->GetCockpitCamera()) { FOV = Ship->GetCockpitCamera()->FieldOfView; }
+		else { FOV = (Ship->IsPlayerFlown() && Ship->GetChaseCamera()) ? Ship->GetChaseCamera()->FieldOfView : Ship->GetChaseCameraFOVNow(); }
+	}
+	const float HalfMin = 0.5f * FMath::Min(SizeX, SizeY);
+	const float Ndc = HalfMin * ReticleScreenScale / (0.5f * SizeX);
+	return FMath::RadiansToDegrees(FMath::Atan(Ndc * FMath::Tan(FMath::DegreesToRadians(FMath::Clamp(FOV, 5.f, 170.f) * 0.5f))));
+}
